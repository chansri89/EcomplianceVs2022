ALTER PROCEDURE [dbo].[AutoMailDataInsertSp] ---run at Web Server
@CmpLinkedToWebServerId bigint,
@InsertResult varchar(64) output
AS
BEGIN TRY
	BEGIN
		-- SET NOCOUNT ON added to prevent extra result sets from
		-- interfering with SELECT statements.
		SET NOCOUNT ON; 
		set @InsertResult = ''
		declare @UpdTextMessage varchar(128)
		Set @UpdTextMessage = 'Please Synchronize Acts. Ensure Internet Connectivity or Service in running Mode or Update Manually'
		declare @OKTextMessage varchar(128)
		Set @OKTextMessage = 'All Acts are Synchronized. Ensure you have Created Activities for them'

		Declare @Result smallint
		Declare @CurrentDownLoadDate datetime
		select @CurrentDownLoadDate = dateadd(minute,330,getutcdate())

---------Insert for Synchronized ACT data Email send only in intervals of MailFrequencyinDays
		insert into tblCompanyLinkedToWebServerMails 
		(CompanyLinkedToWebServerId,MailTo,MailCC,MailText,IsDownLoadedMessage)

		Select tclw.Id as CompanyLinkedToWebServerId, MailTo,MailCC,
		case when isnull(Kount,0) = 0 then @OKTextMessage else @UpdTextMessage end as TxtMsg,
		Case when isnull(Kount,0) = 0 then 1 else 0 end as IsDownloaded
		from tblCompanyLinkedToWebServer tclw
		left Outer join -- Kount=0 means Synchronized, else synchronization to be done.
			(	Select tcl.ClientId,PrevDownLoadDate,LastMailDate,
				case when sum(tamd.Id) is null then 0 else 1 end as Kount from --count(*) as Kount 
				(----identify the last Downloaded date from tblActDownLoadInformation
					Select  tclw.ClientId,max(isnull(tadli.ActRequestDate,'2015-04-01')) as PrevDownLoadDate,
						max(isnull(tclwm.CreatedDate,dateadd(minute,330,getutcdate()))) as LastMailDate
					from tblCompanyLinkedToWebServer tclw
					left outer join tblCompanyLinkedToWebServerMails tclwm on tclwm.CompanyLinkedToWebServerId = tclw.Id  
					left Outer join tblActDownLoadInformation tadli on tclw.ClientId = tadli.ClientId and tadli.IsDownLoaded = 1
					where tclw.id = @CmpLinkedToWebServerId and tclw.IsActive = 1 group by tclw.ClientId
				) tcl -- Check with Act Master for change in Act data using Created and modifieddate of ACT
				left outer join tblActMasterdtl tamd 
					on (tamd.Createddate between PrevDownLoadDate and @CurrentDownLoadDate 
						 or (isnull(tamd.Modifieddate,'2015-04-01') between PrevDownLoadDate and @CurrentDownLoadDate))	
				Group by tcl.ClientId,PrevDownLoadDate,LastMailDate
			) tc on tc.ClientId = tclw.ClientId
		Where tclw.id = @CmpLinkedToWebServerId and tc.Kount=0 and Datediff(dd,tc.LastMailDate,@CurrentDownLoadDate)>=tclw.MailFrequencyinDays

---------Insert for Non Synchronized ACT data Email 
		insert into tblCompanyLinkedToWebServerMails 
		(CompanyLinkedToWebServerId,MailTo,MailCC,MailText,IsDownLoadedMessage)

		Select tclw.Id as CompanyLinkedToWebServerId, MailTo,MailCC,
		case when isnull(Kount,0) = 0 then @OKTextMessage else @UpdTextMessage end as TxtMsg,
		Case when isnull(Kount,0) = 0 then 1 else 0 end as IsDownloaded
		from tblCompanyLinkedToWebServer tclw
		left Outer join -- Kount=0 means Synchronized, else synchronization to be done.
			(	Select tcl.ClientId,PrevDownLoadDate,LastMailDate,
				case when sum(tamd.Id) is null then 0 else 1 end as Kount from --count(*) as Kount 
				(----identify the last Downloaded date from tblActDownLoadInformation
					Select  tclw.ClientId,max(isnull(tadli.ActRequestDate,'2015-04-01')) as PrevDownLoadDate,
						max(isnull(tclwm.CreatedDate,dateadd(minute,330,getutcdate()))) as LastMailDate
					from tblCompanyLinkedToWebServer tclw
					left outer join tblCompanyLinkedToWebServerMails tclwm on tclwm.CompanyLinkedToWebServerId = tclw.Id  
					left Outer join tblActDownLoadInformation tadli on tclw.ClientId = tadli.ClientId and tadli.IsDownLoaded = 1
					where tclw.id = @CmpLinkedToWebServerId and tclw.IsActive = 1 group by tclw.ClientId
				) tcl -- Check with Act Master for change in Act data using Created and modifieddate of ACT
				left outer join tblActMasterdtl tamd 
					on (tamd.Createddate between PrevDownLoadDate and @CurrentDownLoadDate 
						 or (isnull(tamd.Modifieddate,'2015-04-01') between PrevDownLoadDate and @CurrentDownLoadDate))	
				Group by tcl.ClientId,PrevDownLoadDate,LastMailDate
			) tc on tc.ClientId = tclw.ClientId
		Where tclw.id = @CmpLinkedToWebServerId and tc.Kount=1

		set @InsertResult = 'Successful'
	END
END TRY

BEGIN CATCH
	DECLARE @ERR_MSG NVARCHAR(4000),
			@ERR_SEVERITY INT
	SELECT @ERR_MSG = ERROR_MESSAGE(),
		@ERR_SEVERITY = ERROR_SEVERITY()
	RAISERROR(@ERR_MSG,@ERR_SEVERITY,1)
END CATCH
